// app/api/cron/retencion/route.ts — una vez al día. Vercel manda Authorization: Bearer $CRON_SECRET.
import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';
import { enviarCorreo } from '@/lib/enviar-correo';

export const runtime = 'nodejs';

// Los correos de marketing (pagador fantasma, invitaciones, resúmenes) quedan APAGADOS hasta que existan
// bajas funcionales: ruta /baja, lista de bajas consultada antes de cada envío y registro de envíos.
// Ponlo en true solo cuando eso esté listo (docs/retencion/CORREOS.md y PARA-LA-IA-DEL-ALUMNO.md §7).
const MARKETING_ACTIVO = false;

export async function GET(req: Request) {
  if (!process.env.CRON_SECRET || req.headers.get('authorization') !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'no autorizado' }, { status: 401 });
  }
  if (!MARKETING_ACTIVO) {
    return NextResponse.json({ ok: true, enviados: 0, motivo: 'marketing desactivado hasta tener bajas funcionales' });
  }
  const db = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);
  // Círculos con suscripción activa y 14 días sin actividad (vista de supabase/retencion.sql).
  const { data, error } = await db.from('retention_ghost_circles').select('*').limit(200);
  if (error) return NextResponse.json({ error: 'no se pudo leer la retención' }, { status: 500 }); // un fallo no se disfraza de "0 enviados"
  let enviados = 0;
  for (const c of data ?? []) {
    // La vista trae user_id (no correo): búscalo con db.auth.admin.getUserById(c.user_id).
    const { data: u } = await db.auth.admin.getUserById(c.user_id);
    if (!u.user?.email) continue;
    // TODO antes de activarlo: consulta la lista de bajas por correo y registra cada envío para no repetirlo.
    await enviarCorreo('pagador_fantasma', u.user.email, { familiar: 'tu familiar' }, `${process.env.NEXT_PUBLIC_SITE_URL}/baja`);
    enviados++;
  }
  return NextResponse.json({ ok: true, enviados });
}
